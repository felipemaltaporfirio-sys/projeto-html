let jogadorX; let jogadorY; let larguraCarro = 30; let alturaCarro = 50; let inimigos = []; let velocidadeInimigos = 5; let pontos = 0; let gameOver = false; let deslocamentoPista = 0; function setup() { createCanvas(400, 600); reiniciarJogo(); } function draw() { background(50); // Cor do asfalto if (!gameOver) { desenharPista(); atualizarJogador(); desenharJogador(); gerenciarInimigos(); exibirPontuacao(); } else { exibirTelaGameOver(); } } // Desenha as zebras/grama e o movimento contínuo da pista function desenharPista() { noStroke(); // Grama nas laterais fill(34, 139, 34); rect(0, 0, 50, height); rect(350, 0, 50, height); // Linhas brancas no centro da pista stroke(255); strokeWeight(4); deslocamentoPista = (deslocamentoPista + velocidadeInimigos) % 40; for (let y = -40; y < height; y += 40) { line(width / 2, y + deslocamentoPista, width / 2, y + 20 + deslocamentoPista); } noStroke(); } // Movimentação do jogador function atualizarJogador() { if (keyIsDown(LEFT_ARROW) && jogadorX > 55) { jogadorX -= 6; } if (keyIsDown(RIGHT_ARROW) && jogadorX < 345 - larguraCarro) { jogadorX += 6; } } // Desenha o F1 do Jogador (Vermelho) function desenharJogador() { fill(220, 20, 60); // Vermelho F1 rect(jogadorX, jogadorY, larguraCarro, alturaCarro, 5); // Aerofólios fill(20); rect(jogadorX - 5, jogadorY + 5, larguraCarro + 10, 8); rect(jogadorX - 2, jogadorY + alturaCarro - 8, larguraCarro + 4, 8); } // Gera e move os carros adversários function gerenciarInimigos() { let intervaloGeracao = Math.max(25, Math.floor(500 / velocidadeInimigos)); // Cria novos inimigos periodicamente if (frameCount % intervaloGeracao === 0) { let novoX = random(60, 340 - larguraCarro); // Garante que o novo inimigo não nasça colado em outro existente let podeGerar = true; for (let inimigo of inimigos) { if (inimigo.y < alturaCarro * 2) { podeGerar = false; break; } } if (podeGerar) { inimigos.push({ x: novoX, y: -alturaCarro }); } } for (let i = inimigos.length - 1; i >= 0; i--) { inimigos[i].y += velocidadeInimigos; // Desenha o adversário (Azul) fill(30, 144, 255); rect(inimigos[i].x, inimigos[i].y, larguraCarro, alturaCarro, 5); // Detecção de Colisão if ( jogadorX < inimigos[i].x + larguraCarro && jogadorX + larguraCarro > inimigos[i].x && jogadorY < inimigos[i].y + alturaCarro && jogadorY + alturaCarro > inimigos[i].y ) { gameOver = true; } // Passou pelo inimigo: ganha pontos if (inimigos[i].y > height) { inimigos.splice(i, 1); pontos += 10; // Aumenta a velocidade progressivamente if (pontos % 50 === 0) { velocidadeInimigos += 0.5; } } } } function exibirPontuacao() { fill(255); textSize(20); textAlign(LEFT, TOP); text("Pontos: " + pontos, 60, 20); } function exibirTelaGameOver() { fill(0, 180); rect(0, 0, width, height); fill(255, 50, 50); textSize(36); textAlign(CENTER, CENTER); text("FIM DE JOGO!", width / 2, height / 2 - 20); fill(255); textSize(20); text("Pontuação Final: " + pontos, width / 2, height / 2 + 20); textSize(14); text("Pressione 'R' ou clique para reiniciar", width / 2, height / 2 + 60); } function reiniciarJogo() { pontos = 0; velocidadeInimigos = 5; inimigos = []; jogadorX = width / 2 - larguraCarro / 2; jogadorY = height - 80; gameOver = false; } // Controles de Reinício function keyPressed() { if ((key === 'r' || key === 'R') && gameOver) { reiniciarJogo(); } } function mousePressed() { if (gameOver) { reiniciarJogo(); } }
