let jogadorX; let jogadorY; let larguraCarro = 30; let alturaCarro = 50; let inimigos = []; let velocidadeInimigos = 5; let pontos = 0; let gameOver = false; function setup() { createCanvas(400, 600); jogadorX = width / 2 - larguraCarro / 2; jogadorY = height - 80; } function draw() { background(50); // Cor do asfalto if (!gameOver) { desenharPista(); atualizarJogador(); desenharJogador(); gerenciarInimigos(); exibirPontuacao(); } else { exibirTelaGameOver(); } } // Desenha a grama e as faixas centrais da pista function desenharPista() { // Grama nas laterais fill(34, 139, 34); rect(0, 0, 50, height); rect(350, 0, 50, height); // Linhas brancas no centro stroke(255); strokeWeight(4); for (let y = 0; y < height; y += 40) { let linhaY = (y + frameCount * 5) % height; line(width / 2, linhaY, width / 2, linhaY + 20); } noStroke(); } // Movimentação do jogador function atualizarJogador() { if (keyIsDown(LEFT_ARROW) && jogadorX > 55) { jogadorX -= 5; } if (keyIsDown(RIGHT_ARROW) && jogadorX < 345 - larguraCarro) { jogadorX += 5; } } // Desenha o F1 do Jogador (Vermelho) function desenharJogador() { fill(220, 20, 60); // Vermelho F1 rect(jogadorX, jogadorY, larguraCarro, alturaCarro, 5); // Aerofólio dianteiro/traseiro fill(20); rect(jogadorX - 5, jogadorY + 5, larguraCarro + 10, 8); rect(jogadorX - 2, jogadorY + alturaCarro - 8, larguraCarro + 4, 8); } // Gera e move os carros adversários function gerenciarInimigos() { // Cria um novo inimigo periodicamente if (frameCount % 60 === 0) { let xInimigo = random(60, 340 - larguraCarro); inimigos.push({ x: xInimigo, y: -alturaCarro }); } for (let i = inimigos.length - 1; i >= 0; i--) { inimigos[i].y += velocidadeInimigos; // Desenha o adversário (Azul) fill(30, 144, 255); rect(inimigos[i].x, inimigos[i].y, larguraCarro, alturaCarro, 5); // Checa colisão com o jogador if ( jogadorX < inimigos[i].x + larguraCarro && jogadorX + larguraCarro > inimigos[i].x && jogadorY < inimigos[i].y + alturaCarro && jogadorY + alturaCarro > inimigos[i].y ) { gameOver = true; } // Se o inimigo sair da tela, soma pontos if (inimigos[i].y > height) { inimigos.splice(i, 1); pontos += 10; // Aumenta a velocidade progressivamente if (pontos % 50 === 0) { velocidadeInimigos += 0.5; } } } } function exibirPontuacao() { fill(255); textSize(20); textAlign(LEFT, TOP); text("Pontos: " + pontos, 60, 20); } function exibirTelaGameOver() { fill(0, 180); rect(0, 0, width, height); fill(255, 50, 50); textSize(36); textAlign(CENTER, CENTER); text("FIM DE JOGO!", width / 2, height / 2 - 20); fill(255); textSize(20); text("Pontuação Final: " + pontos, width / 2, height / 2 + 20); textSize(14); text("Pressione R para reiniciar", width / 2, height / 2 + 60); } // Reinicia o jogo ao pressionar 'R' function keyPressed() { if ((key === 'r' || key === 'R') && gameOver) { pontos = 0; velocidadeInimigos = 5; inimigos = []; jogadorX = width / 2 - larguraCarro / 2; gameOver = false; } }
